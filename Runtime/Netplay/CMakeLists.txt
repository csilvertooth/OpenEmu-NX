cmake_minimum_required(VERSION 3.20)
project(emu_netplay LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(GNS_ROOT "" CACHE PATH "Valve GNS checkout pinned below")
set(GNS_ARCHIVE "" CACHE FILEPATH "Static archive built from the pinned checkout")
set(DEPS_ROOT "" CACHE PATH "Matching static OpenSSL/protobuf prefix")
set(RUNTIME_LIBRARY "" CACHE FILEPATH "Built libemu_snes.dylib")
execute_process(COMMAND git -C "${GNS_ROOT}" rev-parse HEAD OUTPUT_VARIABLE GNS_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT GNS_COMMIT STREQUAL "d534c19aa760df3fb75fd20db13ba1932b8a5463")
    message(FATAL_ERROR "Unexpected GNS revision")
endif()
foreach(FILE "${GNS_ARCHIVE}" "${DEPS_ROOT}/lib/libprotobuf.a" "${DEPS_ROOT}/lib/libcrypto.a" "${RUNTIME_LIBRARY}")
    if(NOT EXISTS "${FILE}")
        message(FATAL_ERROR "Missing dependency ${FILE}")
    endif()
endforeach()
add_library(emu_runtime SHARED IMPORTED)
set_target_properties(emu_runtime PROPERTIES IMPORTED_LOCATION "${RUNTIME_LIBRARY}")
add_executable(emu-lan session_main.cpp gns_lan.cpp)
target_compile_definitions(emu-lan PRIVATE STEAMNETWORKINGSOCKETS_OPENSOURCE)
target_include_directories(emu-lan PRIVATE "${GNS_ROOT}/include" "${DEPS_ROOT}/include" ../Snes9x)
target_link_libraries(emu-lan PRIVATE emu_runtime "${GNS_ARCHIVE}" "${DEPS_ROOT}/lib/libprotobuf.a" "${DEPS_ROOT}/lib/libcrypto.a" "-framework Security" "-framework CoreFoundation")
set_target_properties(emu-lan PROPERTIES BUILD_RPATH "@loader_path")
add_custom_command(TARGET emu-lan POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${RUNTIME_LIBRARY}" "$<TARGET_FILE_DIR:emu-lan>/libemu_snes.dylib")
add_executable(emu-net-tests tests.cpp)
enable_testing()
add_test(NAME bounded-lockstep-and-wire COMMAND emu-net-tests)

enable_language(OBJCXX)
add_executable(emu-live live.mm multiplayer_save.cpp)
target_compile_options(emu-live PRIVATE -fobjc-arc)
target_compile_definitions(emu-live PRIVATE STEAMNETWORKINGSOCKETS_OPENSOURCE)
target_include_directories(emu-live PRIVATE "${GNS_ROOT}/include" "${DEPS_ROOT}/include" ../Snes9x)
target_link_libraries(emu-live PRIVATE emu_runtime "${GNS_ARCHIVE}" "${DEPS_ROOT}/lib/libprotobuf.a" "${DEPS_ROOT}/lib/libcrypto.a" z "-framework Cocoa" "-framework AVFoundation" "-framework GameController" "-framework QuartzCore" "-framework Security")
set_target_properties(emu-live PROPERTIES BUILD_RPATH "@loader_path")
add_custom_command(TARGET emu-live POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${RUNTIME_LIBRARY}" "$<TARGET_FILE_DIR:emu-live>/libemu_snes.dylib")

add_executable(emu-spectator-tests spectator_tests.cpp)
target_link_libraries(emu-spectator-tests PRIVATE z)
add_test(NAME spectator-av-bundles COMMAND emu-spectator-tests)

add_executable(emu-audio-tests audio_ring_tests.cpp)
add_test(NAME continuous-audio-ring COMMAND emu-audio-tests)

add_executable(emu-rollback-tests rollback_tests.cpp)
target_include_directories(emu-rollback-tests PRIVATE ../Snes9x)
target_link_libraries(emu-rollback-tests PRIVATE emu_runtime)
set_target_properties(emu-rollback-tests PROPERTIES BUILD_RPATH "@loader_path")
add_test(NAME rollback-scheduler-and-audio COMMAND emu-rollback-tests)

add_executable(emu-input-transport-tests input_transport_tests.cpp)
add_test(NAME redundant-input-transport COMMAND emu-input-transport-tests)

add_executable(emu-save-tests multiplayer_save_tests.cpp multiplayer_save.cpp)
target_include_directories(emu-save-tests PRIVATE ../Snes9x "${DEPS_ROOT}/include")
target_link_libraries(emu-save-tests PRIVATE emu_runtime "${DEPS_ROOT}/lib/libcrypto.a")
set_target_properties(emu-save-tests PROPERTIES BUILD_RPATH "@loader_path")
add_test(NAME multiplayer-save-journal COMMAND emu-save-tests)

add_executable(emu-lobby-retry-tests lobby_retry_tests.cpp)
add_test(NAME bounded-lobby-retry COMMAND emu-lobby-retry-tests)

add_executable(emu-access-token-tests access_token_tests.cpp)
add_test(NAME installation-access-token COMMAND emu-access-token-tests)

# Native plugin backend is optional for command-line-only builds.
set(OE_FRAMEWORK_DIR "" CACHE PATH "Built OpenEmu SDK frameworks")
if(OE_FRAMEWORK_DIR)
    add_library(emu_openemu SHARED ../OpenEmu/runtime.mm)
    target_compile_options(emu_openemu PRIVATE -fobjc-arc "-F${OE_FRAMEWORK_DIR}")
    target_link_options(emu_openemu PRIVATE "-F${OE_FRAMEWORK_DIR}")
    target_link_libraries(emu_openemu PRIVATE "-framework Cocoa" "-framework OpenEmuBase")
    set_target_properties(emu_openemu PROPERTIES BUILD_RPATH "${OE_FRAMEWORK_DIR};@loader_path/../../Frameworks;@loader_path/../../../../../Frameworks")
    add_dependencies(emu-live emu_openemu)
endif()

add_executable(emu-multitap-tests multitap_tests.cpp)
target_include_directories(emu-multitap-tests PRIVATE ../Snes9x "${DEPS_ROOT}/include")
target_link_libraries(emu-multitap-tests PRIVATE emu_runtime "${DEPS_ROOT}/lib/libcrypto.a")
set_target_properties(emu-multitap-tests PROPERTIES BUILD_RPATH "@loader_path")
